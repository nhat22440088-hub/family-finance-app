Cấu Hình Kết Nối Supabase Cloud

Đồng bộ cơ sở dữ liệu thời gian thực và lưu trữ danh mục động

1 Tạo Project trên Supabase

Truy cập supabase.com, đăng ký miễn phí và tạo một Project mới (chọn khu vực Singapore gần Việt Nam nhất).

2 Chạy Mã SQL Tạo Bảng (Chỉ 1 Click)

Vào mục SQL Editor -> New query, dán đoạn mã bên dưới và bấm Run:

CREATE TABLE IF NOT EXISTS public.transactions (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    member TEXT NOT NULL,
    type TEXT NOT NULL,
    amount NUMERIC NOT NULL,
    category TEXT NOT NULL,
    note TEXT DEFAULT '',
    date TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.categories (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    type TEXT NOT NULL,
    icon TEXT DEFAULT '📌',
    is_custom BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.transactions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public all" ON public.transactions FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE public.categories ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public all" ON public.categories FOR ALL USING (true) WITH CHECK (true);
